# A private, CPU-only inference DLL. Never links browser code into the engine.
cmake_minimum_required(VERSION 3.24)
project(krypton_translate LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  message(FATAL_ERROR "Use the prepared Chromium clang-cl toolchain on Windows")
endif()
if(NOT DEFINED KRYPTON_ENGINE_SOURCE OR NOT DEFINED KRYPTON_ABI_INCLUDE)
  message(FATAL_ERROR "Run tools/krypton/translation_engine.py first")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(COMPILE_TESTS OFF CACHE BOOL "" FORCE)
set(COMPILE_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(COMPILE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(COMPILE_LIBRARY_ONLY ON CACHE BOOL "" FORCE)
set(COMPILE_PYTHON OFF CACHE BOOL "" FORCE)
set(USE_WASM_COMPATIBLE_SOURCE ON CACHE BOOL "" FORCE)
set(COMPILE_WITHOUT_EXCEPTIONS OFF CACHE BOOL "" FORCE)
set(COMPILE_CUDA OFF CACHE BOOL "" FORCE)
set(USE_MKL OFF CACHE BOOL "" FORCE)
set(USE_DOXYGEN OFF CACHE BOOL "" FORCE)
set(USE_THREADS OFF CACHE BOOL "" FORCE)
set(USE_ONNX_SGEMM ON CACHE BOOL "" FORCE)
set(USE_RUY OFF CACHE BOOL "" FORCE)
set(USE_INTGEMM ON CACHE BOOL "" FORCE)
set(INTGEMM_CPUID_ENVIRONMENT OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(GIT_SUBMODULE OFF CACHE BOOL "" FORCE)
set(BUILD_ARCH x86-64-v3 CACHE STRING "" FORCE)
add_compile_definitions(KRYPTON_TRANSLATE_NO_FILES NOMINMAX
                        _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS EIGEN_MPL2_ONLY)
add_compile_options(/MT /EHsc /guard:cf /Zc:__cplusplus /bigobj /utf-8 /arch:AVX2)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_SUPPORT_JIT OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${KRYPTON_ENGINE_SOURCE}/pcre2" pcre2 EXCLUDE_FROM_ALL)
add_compile_definitions(PCRE2_STATIC)
add_subdirectory("${KRYPTON_ENGINE_SOURCE}/inference" bergamot EXCLUDE_FROM_ALL)

add_library(krypton_translate SHARED src/translate_abi.cc)
target_include_directories(krypton_translate PRIVATE "${KRYPTON_ABI_INCLUDE}")
# Keep diagnostics for owned ABI code strict while treating pinned third-party
# declarations as system headers at this boundary.
foreach(inference_dependency bergamot-translator-source marian intgemm)
  get_target_property(inference_includes ${inference_dependency} INTERFACE_INCLUDE_DIRECTORIES)
  if(inference_includes)
    target_include_directories(krypton_translate SYSTEM PRIVATE ${inference_includes})
  endif()
endforeach()
target_link_libraries(krypton_translate PRIVATE bergamot-translator-source)
target_compile_options(krypton_translate PRIVATE /W4 /WX)
# The pinned inference headers predate clang's newer /W4 diagnostics. These
# narrow compatibility exclusions follow the engine only, never browser/UI code.
foreach(engine_target marian bergamot-translator-source intgemm krypton_translate)
  target_compile_options(${engine_target} PRIVATE -Wno-sign-compare
    -Wno-unused-parameter -Wno-unneeded-internal-declaration -Wno-unused-value
    -Wno-unused-function -Wno-unused-private-field)
endforeach()
target_link_options(krypton_translate PRIVATE /guard:cf /CETCOMPAT /DYNAMICBASE
                    /HIGHENTROPYVA /NXCOMPAT /INCREMENTAL:NO /OPT:REF /OPT:ICF)
set_target_properties(krypton_translate PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

include("${CMAKE_CURRENT_LIST_DIR}/../tools/krypton/compiler_inputs.cmake")
krypton_track_compiler_inputs("${CMAKE_CURRENT_SOURCE_DIR}")
